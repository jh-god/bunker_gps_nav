cmake_minimum_required(VERSION 3.16)
project(bunker_gps_nav_perception)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters)
include_directories(include ${PCL_INCLUDE_DIRS})
add_executable(obstacle_cloud_filter_node src/obstacle_cloud_filter_node.cpp)
ament_target_dependencies(obstacle_cloud_filter_node rclcpp sensor_msgs diagnostic_msgs tf2_ros tf2_sensor_msgs pcl_conversions)
target_link_libraries(obstacle_cloud_filter_node ${PCL_LIBRARIES})
install(TARGETS obstacle_cloud_filter_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_point_filter test/test_point_filter.cpp)
  ament_add_gtest(test_radius_outlier_filter test/test_radius_outlier_filter.cpp)
  target_link_libraries(test_radius_outlier_filter ${PCL_LIBRARIES})
endif()
ament_package()
