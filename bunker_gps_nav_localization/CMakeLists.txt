cmake_minimum_required(VERSION 3.16)
project(bunker_gps_nav_localization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(ublox_msgs REQUIRED)
find_package(robot_localization REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_library(GEOGRAPHIC_LIBRARY Geographic REQUIRED)
include_directories(include)
add_executable(gnss_midpoint_node src/gnss_midpoint_node.cpp)
ament_target_dependencies(gnss_midpoint_node rclcpp sensor_msgs nav_msgs std_msgs diagnostic_msgs ublox_msgs robot_localization tf2 tf2_geometry_msgs)
target_link_libraries(gnss_midpoint_node ${GEOGRAPHIC_LIBRARY})
add_executable(gnss_heading_node src/gnss_heading_node.cpp)
ament_target_dependencies(gnss_heading_node rclcpp sensor_msgs nav_msgs std_msgs diagnostic_msgs ublox_msgs robot_localization tf2 tf2_geometry_msgs)
target_link_libraries(gnss_heading_node ${GEOGRAPHIC_LIBRARY})
add_executable(datum_manager_node src/datum_manager_node.cpp)
ament_target_dependencies(datum_manager_node rclcpp sensor_msgs nav_msgs std_msgs diagnostic_msgs ublox_msgs robot_localization tf2 tf2_geometry_msgs)
target_link_libraries(datum_manager_node ${GEOGRAPHIC_LIBRARY})
install(TARGETS gnss_midpoint_node gnss_heading_node datum_manager_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_gnss test/test_gnss.cpp)
  ament_target_dependencies(test_gnss ublox_msgs)
  target_link_libraries(test_gnss ${GEOGRAPHIC_LIBRARY})
endif()
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
